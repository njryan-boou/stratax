78 Matrix(std::initializer_list<std::initializer_list<T>> list)
79 {
80 std::size_t rows = list.size();
81 std::size_t cols = (rows == 0) ? 0 : list.begin()->size();
82
83 // Ensure all rows have the same length
84 for (const auto& row : list)
85 {
86 if (row.size() != cols)
87 {
88 throw Exceptions::ShapeError("Matrix initializer rows must all have the same number of columns.");
89 }
90 }
91
92 shape_ = stratax::core::Shape({rows, cols}, stratax::core::Shape::allow_zero);
93 strides_ = stratax::core::Strides(shape_);
94 buffer_ = stratax::core::Buffer<T>(
95 core::validation::checked_multiply(rows, cols, "Matrix size overflow"));
96
97 std::size_t index = 0;
98
99 for (const auto& row : list)
100 {
101 for (const auto& value : row)
102 {
103 buffer_[index++] = value;
104 }
105 }
106 }
