79 Matrix(std::initializer_list<std::initializer_list<T>> list)
80 {
81 std::size_t rows = list.size();
82 std::size_t cols = (rows == 0) ? 0 : list.begin()->size();
83
84 // Ensure all rows have the same length
85 for (const auto& row : list)
86 {
87 if (row.size() != cols)
88 {
89 throw Exceptions::ShapeError("Matrix initializer rows must all have the same number of columns.");
90 }
91 }
92
93 shape_ = stratax::core::Shape({rows, cols}, stratax::core::Shape::allow_zero);
94 strides_ = stratax::core::Strides(shape_);
95 buffer_ = stratax::core::Buffer<T>(
96 core::validation::checked_multiply(rows, cols, "Matrix size overflow"));
97
98 std::size_t index = 0;
99
100 for (const auto& row : list)
101 {
102 for (const auto& value : row)
103 {
104 buffer_[index++] = value;
105 }
106 }
107 }
