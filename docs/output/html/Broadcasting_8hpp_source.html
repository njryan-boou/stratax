22inline std::size_t flat_operand_index(
23 std::size_t result_index,
24 const Shape& result_shape,
25 const Shape& operand_shape)
26{
27 std::size_t operand_index = 0;
28 std::size_t operand_stride = 1;
29
30 for (std::size_t result_axis = result_shape.rank(); result_axis-- > 0;)
31 {
32 const std::size_t coordinate =
33 result_index % result_shape[result_axis];
34 result_index /= result_shape[result_axis];
35
36 const std::size_t offset = result_shape.rank() - 1 - result_axis;
37 if (offset >= operand_shape.rank())
38 {
39 continue;
40 }
41
42 const std::size_t operand_axis = operand_shape.rank() - 1 - offset;
43 const std::size_t operand_dimension = operand_shape[operand_axis];
44 const std::size_t operand_coordinate =
45 operand_dimension == 1 ? 0 : coordinate;
46
47 operand_index += operand_coordinate * operand_stride;
48 operand_stride *= operand_dimension;
49 }
50
51 return operand_index;
52}

57inline bool validate_broadcast(
58 const stratax::core::Shape& shape1,
59 const stratax::core::Shape& shape2)
60{
61 const std::size_t result_rank = std::max(shape1.rank(), shape2.rank());
62
63 for (std::size_t offset = 0; offset < result_rank; ++offset)
64 {
65 const std::size_t left =
66 stratax::core::broadcast_detail::dimension_from_right(shape1, offset);
67 const std::size_t right =
68 stratax::core::broadcast_detail::dimension_from_right(shape2, offset);
69
70 if (left != right && left != 1 && right != 1)
71 {
72 return false;
73 }
74 }
75
76 return true;
77}

80inline stratax::core::Shape broadcasted_shape(
81 const stratax::core::Shape& shape1,
82 const stratax::core::Shape& shape2)
83{
84 if (!validate_broadcast(shape1, shape2))
85 {
86 throw Exceptions::BroadcastError("Shapes are not broadcastable");
87 }
88
89 const std::size_t result_rank = std::max(shape1.rank(), shape2.rank());
90 std::vector<std::size_t> result(result_rank);
91
92 for (std::size_t offset = 0; offset < result_rank; ++offset)
93 {
94 const std::size_t left =
95 stratax::core::broadcast_detail::dimension_from_right(shape1, offset);
96 const std::size_t right =
97 stratax::core::broadcast_detail::dimension_from_right(shape2, offset);
98
99 result[result_rank - 1 - offset] = left == 1 ? right : left;
100 }
101
102 return stratax::core::Shape{result};
103}
