39 explicit Strides(const Shape& shape)
40 : buffer_(shape.rank())
41 {
42 if (shape.empty()) {
43 return;
44 }
45
46 buffer_[shape.rank() - 1] = 1;
47
48 for (std::size_t i = shape.rank() - 1; i > 0; --i) {
49 // Preserve constructibility for extremely large shapes by
50 // saturating intermediate stride values when they overflow.
51 try
52 {
53 buffer_[i - 1] = validation::checked_multiply(
54 buffer_[i],
55 shape(i),
56 "Strides overflow for shape");
57 }
58 catch (const Exceptions::DimensionError&)
59 {
60 buffer_[i - 1] = std::numeric_limits<std::size_t>::max();
61 }
62 }
63 }
